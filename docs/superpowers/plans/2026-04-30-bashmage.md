# BashMage Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Textual TUI app that lets users browse/search a TOML-backed bash command library and prints the selected (placeholder-filled) command to stdout on exit.

**Architecture:** Single Python package installed via pipx. `storage.py` owns the TOML loading and `Command` dataclass. Four Textual widgets compose the UI: `SearchBar` (always-focused input), `CommandList` (filtered left panel), `DetailPanel` (right panel), and `PlaceholderModal` (overlay for `{placeholder}` tokens). `__main__.py` redirects Textual's rendering to `/dev/tty` when stdout is piped, then prints the final command after the app exits.

**Tech Stack:** Python 3.11+, Textual ≥ 0.47.0, tomllib (stdlib), pytest, pytest-asyncio, pipx

```python
import re
from dataclasses import dataclass


@dataclass
class Command:
    name: str
    description: str
    command: str

    @property
    def placeholders(self) -> list[str]:
        return list(dict.fromkeys(re.findall(r'\{(\w+)\}', self.command)))

    def fill(self, values: dict[str, str]) -> str:
        result = self.command
        for key, value in values.items():
            result = result.replace(f'{{{key}}}', value)
        return result
```

```python
import re
import tomllib
from dataclasses import dataclass
from pathlib import Path


CONFIG_PATH = Path.home() / ".config" / "bashmage" / "commands.toml"

DEFAULT_COMMANDS = """\
[[commands]]
name = "git-status"
description = "Show working tree status"
command = "git status --short --branch"

[[commands]]
name = "kill-port"
description = "Kill process on a port"
command = "kill -9 $(lsof -t -i:{port})"

[[commands]]
name = "list-ports"
description = "List all listening ports"
command = "lsof -i -P -n | grep LISTEN"
"""


@dataclass
class Command:
    name: str
    description: str
    command: str

    @property
    def placeholders(self) -> list[str]:
        return list(dict.fromkeys(re.findall(r'\{(\w+)\}', self.command)))

    def fill(self, values: dict[str, str]) -> str:
        result = self.command
        for key, value in values.items():
            result = result.replace(f'{{{key}}}', value)
        return result


def load_commands() -> list[Command]:
    if not CONFIG_PATH.exists():
        CONFIG_PATH.parent.mkdir(parents=True, exist_ok=True)
        CONFIG_PATH.write_text(DEFAULT_COMMANDS)
    data = tomllib.loads(CONFIG_PATH.read_text())
    return [Command(**c) for c in data.get("commands", [])]
```

```python
from textual.app import ComposeResult
from textual.message import Message
from textual.widget import Widget
from textual.widgets import Label, ListItem, ListView

from bashmage.storage import Command


class CommandItem(ListItem):
    def __init__(self, command: Command) -> None:
        super().__init__()
        self.command = command

    def compose(self) -> ComposeResult:
        yield Label(f"{self.command.name}\n[dim]{self.command.description}[/dim]")


class CommandList(Widget):
    DEFAULT_CSS = """
    CommandList {
        width: 1fr;
        border-right: solid $primary;
    }
    CommandList ListView {
        height: 1fr;
    }
    """

    class Highlighted(Message):
        def __init__(self, command: Command) -> None:
            super().__init__()
            self.command = command

    class Selected(Message):
        def __init__(self, command: Command) -> None:
            super().__init__()
            self.command = command

    def __init__(self, commands: list[Command]) -> None:
        super().__init__()
        self._all_commands = commands
        self._filtered = list(commands)

    def compose(self) -> ComposeResult:
        yield ListView(*[CommandItem(c) for c in self._all_commands])

    EMPTY_MESSAGE = "(no commands)"
    NO_MATCH_MESSAGE = "(no matches)"

    def filter(self, query: str) -> None:
        q = query.lower()
        self._filtered = [
            c for c in self._all_commands
            if not q or q in c.name.lower() or q in c.description.lower()
        ]
        lv = self.query_one(ListView)
        lv.clear()
        if self._filtered:
            for cmd in self._filtered:
                lv.append(CommandItem(cmd))
        else:
            msg = self.NO_MATCH_MESSAGE if q else self.EMPTY_MESSAGE
            lv.append(ListItem(Label(f"[dim]{msg}[/dim]")))

    def on_list_view_highlighted(self, event: ListView.Highlighted) -> None:
        if isinstance(event.item, CommandItem):
            self.post_message(self.Highlighted(event.item.command))

    def on_list_view_selected(self, event: ListView.Selected) -> None:
        if isinstance(event.item, CommandItem):
            self.post_message(self.Selected(event.item.command))
```

```python
from textual.app import ComposeResult
from textual.containers import Container, Horizontal
from textual.screen import ModalScreen
from textual.widgets import Button, Input, Label, Static

from bashmage.storage import Command


class PlaceholderModal(ModalScreen[str | None]):
    DEFAULT_CSS = """
    PlaceholderModal {
        align: center middle;
    }
    #modal-container {
        width: 60;
        background: $surface;
        border: solid $primary;
        padding: 1 2;
    }
    #modal-title {
        text-style: bold;
        margin-bottom: 1;
    }
    #modal-command {
        color: $text-muted;
        margin-bottom: 1;
    }
    #modal-buttons {
        margin-top: 1;
        align-horizontal: right;
    }
    """

    def __init__(self, command: Command) -> None:
        super().__init__()
        self._command = command

    def compose(self) -> ComposeResult:
        with Container(id="modal-container"):
            yield Static("Fill in placeholders", id="modal-title")
            yield Static(self._command.command, id="modal-command")
            for placeholder in self._command.placeholders:
                yield Label(placeholder.upper())
                yield Input(placeholder=placeholder, id=f"input-{placeholder}")
            with Horizontal(id="modal-buttons"):
                yield Button("Cancel", id="cancel", variant="default")
                yield Button("Confirm", id="confirm", variant="primary")

    def on_mount(self) -> None:
        first = self._command.placeholders
        if first:
            self.query_one(f"#input-{first[0]}", Input).focus()

    def on_button_pressed(self, event: Button.Pressed) -> None:
        if event.button.id == "cancel":
            self.dismiss(None)
        elif event.button.id == "confirm":
            self._try_confirm()

    def on_key(self, event) -> None:
        if event.key == "escape":
            self.dismiss(None)
        elif event.key == "enter":
            self._try_confirm()

    def _try_confirm(self) -> None:
        values: dict[str, str] = {}
        for placeholder in self._command.placeholders:
            widget = self.query_one(f"#input-{placeholder}", Input)
            if not widget.value.strip():
                widget.focus()
                return
            values[placeholder] = widget.value.strip()
        self.dismiss(self._command.fill(values))
```

```python
from textual.app import App, ComposeResult
from textual.binding import Binding
from textual.containers import Horizontal, Vertical
from textual.widgets import Footer

from bashmage.storage import Command
from bashmage.widgets.command_list import CommandList
from bashmage.widgets.detail_panel import DetailPanel
from bashmage.widgets.placeholder_modal import PlaceholderModal
from bashmage.widgets.search_bar import SearchBar


class BashMageApp(App[str | None]):
    CSS = """
    Screen {
        layout: vertical;
    }
    #main {
        layout: horizontal;
        height: 1fr;
    }
    """

    BINDINGS = [
        Binding("up", "list_up", "Up", priority=True),
        Binding("down", "list_down", "Down", priority=True),
        Binding("enter", "select_command", "Select", priority=True),
        Binding("escape", "clear_search", "Clear"),
        Binding("q", "quit_clean", "Quit"),
    ]

    def __init__(self, commands: list[Command]) -> None:
        super().__init__()
        self._commands = commands
        self._highlighted: Command | None = commands[0] if commands else None

    def compose(self) -> ComposeResult:
        yield SearchBar()
        with Horizontal(id="main"):
            yield CommandList(self._commands)
            yield DetailPanel()
        yield Footer()

    def on_search_bar_changed(self, event: SearchBar.Changed) -> None:
        self.query_one(CommandList).filter(event.value)

    def on_command_list_highlighted(self, event: CommandList.Highlighted) -> None:
        self._highlighted = event.command
        self.query_one(DetailPanel).show(event.command)

    async def on_command_list_selected(self, event: CommandList.Selected) -> None:
        await self._activate(event.command)

    def action_list_up(self) -> None:
        from textual.widgets import ListView
        self.query_one(ListView).action_cursor_up()

    def action_list_down(self) -> None:
        from textual.widgets import ListView
        self.query_one(ListView).action_cursor_down()

    async def action_select_command(self) -> None:
        if self._highlighted is not None:
            await self._activate(self._highlighted)

    def action_clear_search(self) -> None:
        self.query_one(SearchBar).clear()

    def action_quit_clean(self) -> None:
        self.exit(None)

    async def _activate(self, command: Command) -> None:
        if command.placeholders:
            result = await self.push_screen_wait(PlaceholderModal(command))
            if result is not None:
                self.exit(result)
        else:
            self.exit(command.command)
```

```python
import sys
from bashmage.app import BashMageApp
from bashmage.storage import load_commands


def main() -> None:
    try:
        commands = load_commands()
    except Exception as e:
        print(f"bashmage: error loading commands: {e}", file=sys.stderr)
        sys.exit(1)

    output_stream = sys.stdout
    if not sys.stdout.isatty():
        tty = open("/dev/tty", "w")
        sys.stdout = tty
        sys.stderr = tty

    app = BashMageApp(commands)
    result = app.run()

    sys.stdout = output_stream

    if result is not None:
        print(result)
        sys.exit(0)
    else:
        sys.exit(1)


if __name__ == "__main__":
    main()
```
